	/***** Alloys: one recipe at the furnace, smelting and casting together, like the game's own smelting
	   (entities/blacksmith/craftRecipes/recipes_blacksmith_furnaces_i.txt): crucible, tongs and ingot mould are kept,
	   charcoal is burnt. Tags = Furnace: the Smelting Furnace and the Blast Furnace. Nothing to learn. The shotgun may
	   be broken but must be empty and bare (SilencerMoon.recipeTestShotgun); the recipe tooltip says so. *****/

	craftRecipe SilencerMoon_SmeltCopperSteel
	{
		time = 200,
		Tags = Furnace,
		category = Blacksmithing,
		NeedToBeLearn = false,
		SkillRequired = Blacksmith:5,
		xpAward = Blacksmith:40,
		Tooltip = Tooltip_SilencerMoon_Smelt,
		OnTest = SilencerMoon.recipeTestShotgun,
		inputs
		{
			item 1 [Base.Shotgun;Base.JS3T_Shotgun;Base.ShotgunSawnoff] mode:destroy flags[AllowDestroyedItem],
			item 2 [Base.CopperScrap],
			item 1 [Base.SmallSheetMetal],
			item 1 [Base.CeramicCrucible] mode:keep flags[IsEmpty],
			item 1 tags[base:crudetongs;base:tongs] mode:keep flags[MayDegradeLight],
			item 8 tags[base:charcoal],
			item 1 [Base.ClayIngotMold;Base.IronIngotMold;Base.SteelIngotMold] mode:keep,
		}
		outputs
		{
			item 4 SilencerMoon.CopperSteelIngot,
		}
	}
